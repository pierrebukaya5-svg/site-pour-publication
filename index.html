const express = require('express'); const nodemailer = require('nodemailer'); const path = require('path'); const app = express(); // Middleware app.use(express.json()); app.use(express.urlencoded({ extended: true })); // Configuration du transporteur d'e-mails pour l'administrateur const transporter = nodemailer.createTransport({ service: 'gmail', auth: { user: process.env.ADMIN_EMAIL, // Votre adresse Gmail d'administrateur pass: process.env.EMAIL_APP_PASSWORD // Votre mot de passe d'application Google } }); // ========================================== // 1. ROUTE BACKEND (API d'enregistrement) // ========================================== app.post('/api/admin/enregistrer-professeur', async (req, res) => { try { const { nom, postnom, matiere } = req.body; if (!nom || !postnom || !matiere) { return res.status(400).json({ success: false, message: "Tous les champs doivent être remplis." }); } // Génération d'un code d'accès unique à 6 caractères const codeAcces = Math.random().toString(36).substring(2, 8).toUpperCase(); // TODO : Insérez ici votre code d'enregistrement dans la base de données (ex: MongoDB / PostgreSQL) // Configuration de l'e-mail envoyé UNIQUEMENT à l'administrateur const mailOptions = { from: process.env.ADMIN_EMAIL, to: process.env.ADMIN_EMAIL, // Envoyé dans votre propre compte Gmail d'admin subject: `[Portail RDC] Code d'accès - Prof. ${nom} ${postnom}`, text: `Bonjour Administrateur,\n\nUn nouveau professeur a été enregistré manuellement :\n\n- Nom : ${nom}\n- Post-nom / Prénom : ${postnom}\n- Matière : ${matiere}\n- Code d'accès généré : ${codeAcces}\n\nVeuillez le transmettre au professeur en toute sécurité.` }; // Envoi de l'e-mail await transporter.sendMail(mailOptions); res.status(200).json({ success: true, message: "Professeur enregistré avec succès ! Le code d'accès a été envoyé dans votre boîte Gmail." }); } catch (error) { console.error("Erreur serveur :", error); res.status(500).json({ success: false, message: "Erreur interne lors de l'enregistrement." }); } }); // ========================================== // 2. INTERFACE HTML & FRONTEND INTÉGRÉS // ========================================== app.get('/', (req, res) => { res.send(` Portail Scolaire RDC - Administration

Enregistrer un Professeur

`); }); // Lancement du serveur const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Serveur actif sur le port ${PORT}`); });
