const express = require('express');
const fs = require('fs');
const app = express();

app.use(express.json());
app.use(express.static('.'));

let data = { eleves: [], profs: [] };
try { data = JSON.parse(fs.readFileSync('donnees.json','utf8')); } catch(e){}

app.post('/api/admin/prof', (req, res) => {
  console.log("Reçu prof:", req.body);
  const nom = req.body.nom || req.body.nomProf || req.body.Nom || "Prof Inconnu";
  const code = req.body.code || req.body.codeAcces || "1234";
  data.profs.push({ nom, code, date: new Date() });
  fs.writeFileSync('donnees.json', JSON.stringify(data, null, 2));
  res.json({ success: true, message: "Prof créé" });
});

app.post('/api/admin/eleve', (req, res) => {
  console.log("Reçu élève:", req.body);
  const nom = req.body.nom || req.body.nomEleve || "Eleve Inconnu";
  data.eleves.push({ nom, ...req.body, matricule: "RDC"+Date.now().toString().slice(-5) });
  fs.writeFileSync('donnees.json', JSON.stringify(data, null, 2));
  res.json({ success: true, message: "Elève créé" });
});

app.get('/', (req,res) => res.sendFile(__dirname + '/index.html'));

const PORT = process.env.PORT || 10000;
app.listen(PORT, ()=> console.log("Portail RDC OK sur "+PORT));
