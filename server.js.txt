const express = require('express');
const app = express();
app.use(express.json());
app.use(express.static('.'));

let eleves = [];
let profs = [];

app.post('/api/admin/prof', (req, res) => {
  try {
    const { nom, code } = req.body; // Maintenant on prend bien nom et code comme sur ta photo
    if(!nom || !code) return res.json({success: false, message: "Nom et code obligatoires"});
    profs.push({ nom, code });
    console.log("Prof créé:", nom, code);
    return res.json({ success: true });
  } catch(e) {
    console.log(e);
    return res.json({ success: false, message: e.message });
  }
});

app.post('/api/admin/eleve', (req, res) => {
  try {
    const { nom, section, tuteur } = req.body;
    const matricule = "RDC" + Math.floor(10000 + Math.random()*90000);
    eleves.push({ nom, section, matricule });
    console.log("Eleve créé:", nom);
    return res.json({ success: true, matricule });
  } catch(e) {
    return res.json({ success: false, message: e.message });
  }
});

app.get('/api/eleves', (req,res)=> res.json(eleves));

const PORT = process.env.PORT || 10000;
app.listen(PORT, () => console.log("Serveur OK sur " + PORT));
